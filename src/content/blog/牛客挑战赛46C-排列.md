题目描述：  
有个1$\sim$n的随机排列的数组，且其含k个超级逆序对，问你能猜对该数组的概率是多少。

数据范围：  
$n,k\leq 500$

题解：  
一道DP题，不算难。  
考虑设$dp[i][j][p]$为$1\sim i$，且i在第p位，有j个超级逆序对的方案数。  
则若加入i，除了i-1，其余的只要在i后均能形成超级逆序对，且不存在在i前与i形成超级逆序对的。  
考虑将i插入到$1\sim i-1$的不同位置，并且讨论和i-1的先后顺序即可。

-   如果i-1在p后，则方案数为$\sum\_{l=p}^{l=i-1}{dp[i-1][j-i+p+1][l]}$
-   如果i-1在p前，则方案数为$\sum\_{l=1}^{l=p-1}{dp[i-1][j-i+p][l]}$

即$dp[i][j][p]=\sum\_{l=p}^{l=i-1}{dp[i-1][j-i+p+1][l]}+\sum\_{l=1}^{l=p-1}{dp[i-1][j-i+p][l]}$

```c++
#include <bits/stdc++.h>
using namespace std;
const int N=505,mod=998244353;
int dp[2][N][N],s[2][N][N];
int n,k;
inline int add(int a,int b){return a+b>=mod?a+b-mod:a+b;}
inline int del(int a,int b){return a<b?a-b+mod:a-b;}
inline int mul(int a,int b){return a*1ll*b%mod;}
inline int qpow(int a,int b){
    int s=1;
    while(b){
        if(b&1)s=mul(s,a);
        a=mul(a,a);
        b>>=1;
    }return s;
}
int main(){
    scanf("%d%d",&n,&k);
    dp[0][0][1]=1;
    s[0][0][1]=1;
    int now=0,las=1;
    for(int i=2;i<=n;++i){
        swap(now,las);
        for(int j=0;j<=k;++j){
             for(int p=1;p<=i;++p){
                 dp[now][j][p]=0;
                 if(j-i+p+1>=0)dp[now][j][p]=del(s[las][j-i+p+1][i-1],s[las][j-i+p+1][p-1]);
                 if(j-i+p>=0)dp[now][j][p]=add(dp[now][j][p],s[las][j-i+p][p-1]);
                 s[now][j][p]=add(s[now][j][p-1],dp[now][j][p]);
             }
        }
    }
    int ans=0;
    for(int i=1;i<=n;++i)ans=add(ans,dp[now][k][i]);
    printf("%d\n",qpow(ans,mod-2));
    return 0;
}
```
