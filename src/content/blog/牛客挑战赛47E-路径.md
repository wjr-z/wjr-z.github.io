## 题目描述

给定一棵树，求所有经过点数大于等于2的无向路径中长度第k小的。

[题目链接](https://ac.nowcoder.com/acm/contest/10743/E)

用点分治先预处理出长度为 i 的路径数 ans\[i\] 。  
假设当前层重心是 x ，设 a\[i\] 表示当前已经处理的所有子树到 x 路径长度为 i 的路径数，b\[i\] 表示正在处理的子树到 x 路径长度为 i 的路径数，则有

$$

ans[i]=\sum_{j=0}^{i}a[j]*b[i-j]

$$

然后将当前处理子树合并到已处理子树上，两数组相加即可。  
关于求这个卷积，如果暴力求，很可能会TLE，如果没有TLE说明数据比较水，反正复杂度肯定是不行的。  
可以用FFT或者NTT来求这个卷积。  
不过就算这样了其实还是很可能会TLE。  
假设当前根有 n-1 棵子树，每颗子树仅一个点，且第一个子树和当前根节点连的边为 100000 ，这样的话会被卡到 $O(n* z)$。  
  
因此需要将当前根节点的所有子树按照其最长路径从小到大排序后再依次合并，类似于启发式合并，复杂度会降低到$O(z\\,logz\\,logn)$  
  
算上点分治复杂度，总复杂度为 $O(z\\,logz\\,logn+n\\,logn)$  
  

```c++
#include <bits/stdc++.h>
using namespace std;
struct Complex {
    double x, y;
    Complex(double a = 0, double b = 0) {
        x = a;
        y = b;
    }
};
inline Complex operator+(Complex a, Complex b) { return Complex(a.x + b.x, a.y + b.y); }
inline Complex operator-(Complex a, Complex b) { return Complex(a.x - b.x, a.y - b.y); }
inline Complex operator*(Complex a, Complex b) { return Complex(a.x * b.x - a.y * b.y, a.x * b.y + a.y * b.x); }
const int N = 1 << 18;
const double PI = acos(-1);
int rev[N];
inline void get_rev(int s) {
    int n = 1 << s;
    for (int i = 0; i < n; ++i)
        rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (s - 1));
}
inline void FFT(Complex* a, int n, int p) {
    for (int i = 0; i < n; ++i)
        if (i < rev[i])
            swap(a[i], a[rev[i]]);
    for (int i = 2, mid = 1; i <= n; i <<= 1, mid <<= 1) {
        Complex wn(cos(2 * PI / i), p * sin(2 * PI / i));
        for (int j = 0; j < n; j += i) {
            Complex w(1, 0);
            for (int k = j; k < j + mid; ++k) {
                Complex x = a[k], y = w * a[k + mid];
                a[k] = x + y, a[k + mid] = x - y;
                w = w * wn;
            }
        }
    }
    if (p == -1)
        for (int i = 0; i < n; ++i)
            a[i].x /= n;
}
const int M = 5e4 + 10, H = 2e5 + 10;
long long ans[H];
Complex F[N], G[N];
inline void Merge(int* a, int* b, int n, int m) {
    int len = n + m, bit = 0, s = 1;
    while (s <= len)++bit, s <<= 1;
    for (int i = 0; i < s; ++i)
        F[i].x = a[i], G[i].x = b[i], F[i].y = G[i].y = 0;
    get_rev(bit);
    FFT(F, s, 1), FFT(G, s, 1);
    for (int i = 0; i < s; ++i)
        F[i] = F[i] * G[i];
    FFT(F, s, -1);
    for (int i = 0; i <= len; ++i)
        ans[i] += (int)(F[i].x + 0.5);
}

int dp[M], siz[M], d[M], nxt[M << 1], to[M << 1], cost[M << 1], tot;
bool vis[M];
int Siz, rot;
inline void ins(int a, int b, int c) {
    to[++tot] = b;
    nxt[tot] = d[a];
    d[a] = tot;
    cost[tot] = c;
}
inline void get_rt(int x, int f) {
    siz[x] = 1; dp[x] = 0;
    for (int i = d[x]; i; i = nxt[i]) {
        int u = to[i];
        if (u == f || vis[u])continue;
        get_rt(u, x);
        siz[x] += siz[u];
        dp[x] = max(dp[x], siz[u]);
    }
    dp[x] = max(dp[x], Siz - siz[x]);
    if (dp[x] < dp[rot])rot = x;
}
int a[N], b[N];
int al, bl;
int stk[M], top;
inline void dfs(int x, int f, int v) {
    ++b[v];
    for (int i = d[x]; i; i = nxt[i]) {
        int u = to[i];
        if (u == f || vis[u])continue;
        dfs(u, x, v + cost[i]);
    }
}
int Max;
struct node {
    int x, max_dep, cost;
    inline bool operator<(const node& other)const {
        return max_dep < other.max_dep;
    }
}st[M];
int cnt;
inline int get_dep(int x, int f, int dep) {
    int c = dep;
    for (int i = d[x]; i; i = nxt[i]) {
        int u = to[i];
        if (u == f || vis[u])continue;
        c = max(c, get_dep(u, x, dep + cost[i]));
    }
    return c;
}
inline void calc(int x) {
    a[0] = 1; al = 0;
    cnt = 0;
    for (int i = d[x]; i; i = nxt[i]) {
        int u = to[i];
        if (vis[u])continue;
        ++cnt;
        st[cnt].x = u, st[cnt].max_dep = get_dep(u, x, cost[i]);
        st[cnt].cost = cost[i];
    }
    sort(st + 1, st + 1 + cnt);
    for (int i = 1; i <= cnt; ++i) {
        int u = st[i].x;
        bl = st[i].max_dep;
        dfs(u, x, st[i].cost);
        Merge(a, b, al, bl);
        Max = max(Max, al + bl);
        al = max(al, bl);
        for (int j = 0; j <= al; ++j)
            a[j] += b[j], b[j] = 0;
    }
    for (int i = 0; i <= al; ++i)
        a[i] = 0;
}
inline void solve(int x, int s) {
    rot = 0; Siz = s;
    get_rt(x, 0);
    x = rot;
    vis[x] = 1; calc(x);
    for (int i = d[x]; i; i = nxt[i]) {
        int u = to[i];
        if (vis[u])continue;
        solve(u, siz[u] < siz[x] ? siz[u] : s - siz[x]);
    }
}
int n, q;
int main() {
    scanf("%d%d", &n, &q);
    dp[0] = n + 1;
    int u, v, w;
    for (int i = 1; i < n; ++i) {
        scanf("%d%d%d", &u, &v, &w);
        ins(u, v, w);
        ins(v, u, w);
    }
    solve(1, n);
    for (int i = 1; i <= Max; ++i)
        ans[i] += ans[i - 1];
    long long k;
    while (q--) {
        scanf("%lld", &k);
        int L = 0, R = Max, Ans = 0;
        while (L <= R) {
            int mid = L + R >> 1;
            if (ans[mid] < k)L = mid + 1;
            else R = mid - 1, Ans = mid;
        }
        printf("%d\n", Ans);
    }
    return 0;
}
```
