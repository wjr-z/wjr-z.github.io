这题其实就是最小斯坦纳树的模板题…  
真的，只要会最小斯坦纳树，这题基本秒解。  
将所有相同宝石当做一个点，然后将x,y也算作必选即可。  
之后就是套最小斯坦纳树模板了。  
不过其实也可以不用这样啦，直接将所有相同宝石的点连一个对应宝石的新增点，边权为0，然后初始化的时候，第i个点如果宝石种类是p，令f\[i\]\[1<<(p-1)\]=0即可。

参考代码

```c++
#include <bits/stdc++.h>
using namespace std;
const int N=120,M=1<<10,inf=1e9+7;
int d[N],nxt[M+2*N],to[M+2*N],cost[M+2*N],tot;
int n,m,k;
int x,y,cnt;
int f[N][M],dis[N];
bool in[N];
queue<int>q;
inline void ins(int a,int b,int c){
    to[++tot]=b;
    nxt[tot]=d[a];
    d[a]=tot;
    cost[tot]=c;
}
inline void spfa(int S){
    for(int i=1;i<=n;++i){
        dis[i]=f[i][S];
        if(dis[i]<inf)q.push(i),in[i]=1;
    }
    while(!q.empty()){
        int x=q.front();
        q.pop();
        in[x]=0;
        for(int i=d[x];i;i=nxt[i]){
            int u=to[i];
            if(dis[u]>dis[x]+cost[i]){
                dis[u]=dis[x]+cost[i];
                if(!in[u])in[u]=1,q.push(u);
            }
        }
    }
    for(int i=1;i<=n;++i)f[i][S]=dis[i];
}
inline void Stenier(){
    cnt=1<<(k+2);
    for(int i=1;i<=n;++i)f[i][0]=0;
    for(int S=0;S<cnt;++S){
        for(int i=1;i<=n;++i)
            for(int T=S&(S-1);T;T=(T-1)&S)
                f[i][S]=min(f[i][S],f[i][T]+f[i][S^T]);
        spfa(S);
    }
}
int main(){
    scanf("%d%d%d%d%d",&n,&m,&k,&x,&y);
    memset(f,0x3f,sizeof(f));
    int p;
    for(int i=1;i<=n;++i){
        scanf("%d",&p);
        if(i==x)f[i][1<<k]=0;
        else if(i==y)f[i][1<<(k+1)]=0;
        else{
            if(!p)continue;
            f[i][1<<(p-1)]=0;
            ins(i,n+p,0);
            ins(n+p,i,0);
        }
    }n+=k;
    int u,v,w;
    for(int i=1;i<=m;++i){
        scanf("%d%d%d",&u,&v,&w);
        if(u==v)continue;
        ins(u,v,w),ins(v,u,w);
    }
    Stenier();
    int ans=inf;
    for(int i=1;i<=n;++i)ans=min(ans,f[i][cnt-1]);
    printf("%d\n",ans);
    return 0;
}
```

感觉这样子的话如果去掉相同宝石可以传送，然后去掉建额外边，好像也是可以的？  
因为其实求最小斯坦纳树除了初始化之外，之后与那k个点无关了。
