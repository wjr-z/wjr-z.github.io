## 前言

这题我居然觉得难点是回溯…

## 题解

这题其实和最小斯坦纳树模板很像，只是将边权改为了点权，而斯坦纳树通过 dp 方式转移的时候根节点 i 的点权会被算两次，因此减去一次即可。  
然后就是套最小斯坦纳树板子了，不过这个回溯着实坑死我了…  
其实就是记下每个 dp 状态是由什么转移过来的即可，最后从终止状态慢慢往回推，只要是回溯到的点都打上标记，代表这个位置安排了志愿者。

代码如下：  

```c++
#include <bits/stdc++.h>
using namespace std;
const int N = 11, M = 1 << 10, inf = 1e9 + 7;
int n, m, c;
struct node {
    int x, s;
    node(int a = 0, int b = 0) {
        x = a;
        s = b;
    }
}pre[N * N][M];
int d[N * N], nxt[N * N * 4], to[N * N * 4], w[N * N], id[N * N], tot;
int maxn, cnt;
inline void ins(int a, int b) {
    to[++tot] = b;
    nxt[tot] = d[a];
    d[a] = tot;
}
int dp[N * N][M], dis[N * N];
queue<int>q;
bool in[N * N];
inline void spfa(int s) {
    for (int i = 1; i <= c; ++i) {
        dis[i] = dp[i][s];
        if (dis[i] < inf)in[i] = 1, q.push(i);
    }
    while (!q.empty()) {
        int x = q.front();
        q.pop();
        in[x] = 0;
        for (int i = d[x]; i; i = nxt[i]) {
            int u = to[i];
            if (dis[u] > dis[x] + w[u]) {
                dis[u] = dis[x] + w[u];
                pre[u][s] = node(x, s);
                if (!in[u])in[u] = 1, q.push(u);
            }
        }
    }
    for (int i = 1; i <= c; ++i)dp[i][s] = dis[i];
}inline void work() {
    maxn = 1 << cnt;
    for (int i = 1; i <= c; ++i)
        dp[i][0] = 0;
    for (int S = 1; S < maxn; ++S) {
        for (int i = 1; i <= c; ++i)
            for (int T = (S - 1) & S; T; T = (T - 1) & S) {
                int v = dp[i][T] + dp[i][S ^ T] - w[i];
                if (v < dp[i][S]) {
                    dp[i][S] = v;
                    pre[i][S] = node(i, T);
                }
            }
        spfa(S);
    }
}int py[4][2] = { {0,1},{1,0},{0,-1},{-1,0} };
inline int get(int i, int j) { return (i - 1) * m + j; }
bool ans[N * N];
inline void dfs(int x, int s) {
    if (!x)return;
    ans[x] = true;
    node p = pre[x][s];
    dfs(p.x, p.s);
    if (p.x == x)dfs(p.x, p.s ^ s);
}
int main() {
    memset(dp, 0x3f, sizeof(dp));
    scanf("%d%d", &n, &m);
    int p;
    int root;
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            int x = ++c;
            scanf("%d", &p);
            w[x] = p;
            if (!p) {
                id[x] = cnt++;
                dp[x][1 << id[x]] = 0;
                root = x;
                pre[x][1 << id[x]] = node(0, 0);
            }
            else {
                id[x] = -1;
            }
        }
    }
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            for (int k = 0; k < 4; ++k) {
                int x = i + py[k][0], y = j + py[k][1];
                if (x<1 || y<1 || x>n || y>m)continue;
                int a = get(i, j), b = get(x, y);
                ins(a, b);
            }
    work();
    printf("%d\n", dp[root][maxn - 1]);
    dfs(root, maxn - 1);
    for (int i = 1; i <= n; ++i) {
        for (int j = 1; j <= m; ++j) {
            int x = get(i, j);
            if (id[x] == -1) {
                if (ans[x])printf("o");
                else printf("_");
            }
            else {
                printf("x");
            }
        }
        printf("\n");
    }
    return 0;
}
```
