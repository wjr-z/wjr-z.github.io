此题如果是栈的形式，则非常好维护DP  
因为删除栈顶DP不变，栈顶增加元素只用$O(p)$修改DP  
  
因此就用到到了**双栈模拟队列**

## 双栈模拟队列

设 F , S 分别是队尾和队首的栈，支持 push（在队尾插入），pop（在队首弹出）操作：

-   push：插入到栈 F 中。
-   pop：如果 S 非空，让 S 弹栈；否则把 F 的元素倒过来压到 S 中（其实就是一个一个弹出插入，做完后是首位颠倒的），然后再让 S 弹栈。

容易证明，每个元素只会进入/转移/弹出一次，均摊复杂度$O(1)$ 。

## 双栈模拟双端队列

与双栈模拟队列相似，仅pop改变

-   pop时，若栈为空，将另一个栈（设元素大小为siz)的前$\frac{siz+1}{2}$倒着压入该栈并弹出一个  
    这样到下一次pop时为空最少需要$\frac{siz}{2}$次pop  
      
    这样的话均摊复杂度是$O(1)$的

## 题解

设 $f[i,j]$ 表示前 i 个二元组，余数为 j 的最大价值

$$

f[i][j]=max(f[i-1][j],f[i-1][(j-w_i+p)\%p]+v_j)

$$

用双栈模拟队列即可很容易的分别维护队首和队尾的 f  
询问可用单调队列优化

```c++
#include <bits/stdc++.h>
using namespace std;
struct node {
    int w,v;
    node(int a = 0, int b = 0) {
        w=a,v=b;
    }
};
const int N=5e4+10,M=505;
int m,p;
inline int _(int x) { return (x + p) % p; }
struct stk {
    int dp[N][M], top;
    node st[N];
    stk() {
        for(int i=1;i<500;++i)
            dp[0][i]=-1;
    }
    void update() {
        for (int i = 0; i < p; ++i) {
            dp[top][i]=dp[top-1][i];
            int k=_(i-st[top].w);
            if(~dp[top-1][k])
                dp[top][i]=max(dp[top][i],dp[top-1][k]+st[top].v);
        }
    }
}a[2];
void push(int o,node x) {
    a[o].st[++a[o].top]=x;
    a[o].update();
}
void pop(int o) {
    if (a[o].top) {
        --a[o].top;
        return;
    }int k=!o;
    int mid=(a[k].top+1)>>1,top=a[k].top;
    for(int i=mid;i>=1;--i)push(o,a[k].st[i]);
    a[k].top=0;
    for(int i=mid+1;i<=top;++i)push(k,a[k].st[i]);
    --a[o].top;
}
int q[N],ql,qr;
int query(int l,int r) {
    ql=1,qr=0;
    int k1=a[0].top,k2=a[1].top;
    for (int i = l - p + 1; i <= r - p + 1; ++i) {
        int x=a[1].dp[k2][_(i)];
        while(ql<=qr&&a[1].dp[k2][q[qr]]<=x)--qr;
        q[++qr]=_(i);
    }
    int ans=-1;
    for (int i = p - 1; i >= 0; --i) {
        if(ql<=qr&&~a[0].dp[k1][i]&&~a[1].dp[k2][q[ql]])
            ans=max(ans,a[0].dp[k1][i]+a[1].dp[k2][q[ql]]);
        if(ql<=qr&&_(l-i)==q[ql])++ql;
        int x=a[1].dp[k2][_(r-i+1)];
        while(ql<=qr&&a[1].dp[k2][q[qr]]<=x)--qr;
        q[++qr]=_(r-i+1);
    }return ans;
}
int main() {
    int kase;
    scanf("%d",&kase);
    scanf("%d%d",&m,&p);
    char s[10];
    int v,w;
    while (m--) {
        scanf("%s%d%d",s,&w,&v);
        int o=s[1]=='F'?0:1;
        if(s[0]=='I')push(o,node(_(w),v));
        if(s[0]=='D')pop(o);
        if(s[0]=='Q')
            printf("%d\n",query(w,v));
    }
    return 0;
}
```
