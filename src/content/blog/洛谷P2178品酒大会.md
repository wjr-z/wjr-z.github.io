字符串好题啊！

题目描述见原题。

这题可以用后缀数组SA做。

我们先常规的后缀数组求出sa、height、rk数组，然后按照rank的顺序从1到n跑。

我们不妨求出对于每个点i，相似度为height\[i\]的所有点对，这样其实就是找到一个rank比rank\[i\]小的height比height\[i\]小的最大的位置，和找到一个rank比rank\[i\]大的height比height\[i\]小的位置。

这样对于其中所有$(j,k),rank[j]\leq i,i<rank[k]$的点对都是相似度为0~height\[i\]的，然后用差分即可，这样很容易就能求出任意相似度的点对数，并且可以通过预处理然后RMQ求出最大美味度。  
不过这个方法呢，需要找到每个点比他height小的两个点，多了一个log n，这题时间卡的还比较紧…(貌似可以用队列优化吧)

第二种方法就是按照rank从小到大跑，考虑用一个队列存相似度相同(为与当前点的相似度)的每一段，每一段记录这一段长度和这一段a的最大和最小值

现在考虑答案贡献问题：

我们从rank从小到大跑，当前到了i，我们将所有相似度不小于height\[i\]的进行答案计算，设这一段为\[L,R\]，和i-1相似度为k，则\[L,R\]中所有点到\[R+1,i-1\]所有点相似度最大为k，通过差分更新第一个答案，通过记下\[R+1,i-1\]中a的最大和最小值更新第二个答案。

对于加入第i个点，首先就是为了符合相似度递增，更新下，然后加入第i个点到队列即可。

注意些细节！

```c++
#include <bits/stdc++.h>
using namespace std;
const int N=3e5+10;
int sa[N],tax[N],rk[N],tp[N];
int n,m;
char s[N];
int he[N];
inline void suffix_sort(){
    memset(tax+1,0,m<<2);
    for(int i=1;i<=n;++i)++tax[rk[i]];
    for(int i=1;i<=m;++i)tax[i]+=tax[i-1];
    for(int i=n;i>=1;--i)sa[tax[rk[tp[i]]]--]=tp[i];
}inline void qsort(){
    m=26;
    for(int i=1;i<=n;++i)rk[i]=s[i]-'a'+1,tp[i]=i;
    suffix_sort();
    for(int i=1,p=0;p<n;m=p,i<<=1){
        p=0;
        for(int j=1;j<=i;++j)tp[++p]=n-i+j;
        for(int j=1;j<=n;++j)
            if(sa[j]>i)
                tp[++p]=sa[j]-i;
        suffix_sort(),swap(tp,rk);
        rk[sa[1]]=p=1;
        for(int j=2;j<=n;++j)rk[sa[j]]=(tp[sa[j]]==tp[sa[j-1]]&&tp[sa[j]+i]==tp[sa[j-1]+i])?p:++p;
    }
}inline void get_height(){
    int j,k=0;
    for(int i=1;i<=n;++i){
        if(k)--k;
        j=sa[rk[i]-1];
        while(s[i+k]==s[j+k])++k;
        he[rk[i]]=k;
    }
}inline void init(){qsort();get_height();}
int maxn[N],minn[N],stk[N],len[N],a[N],top;
long long sum[N],Max[N];
const int inf=1e9;
const long long INF=1e18+7;
inline void insert(int pos){
    int h=he[pos],cnt=0,H,p=-inf,q=inf;
    int now,v=a[sa[pos]];
    while(top&&(now=stk[top])>=h){ 
        cnt+=len[top];
        if(now^inf)sum[now+1]+=len[top]*1ll*(n-pos+1),Max[now]=max(Max[now],max(maxn[top]*1ll*p,minn[top]*1ll*q));
        else sum[0]+=len[top]*1ll*(n-pos+1);
        p=max(p,maxn[top]),q=min(q,minn[top]);
        --top;
    }if(cnt){
        stk[++top]=he[pos];len[top]=cnt;maxn[top]=p,minn[top]=q;
        sum[h+1]-=len[top]*1ll*(n-pos+1);Max[h]=max(Max[h],max(v*1ll*maxn[top],v*1ll*minn[top]));
    }
}
int main(){
    scanf("%d%s",&n,s+1);
    init();
    Max[0]=-INF;
    for(int i=1;i<=n;++i)scanf("%d",&a[i]),Max[i]=-INF;
    he[n+1]=0;
    for(int i=1;i<=n+1;++i){
        insert(i);
        stk[++top]=inf;
        len[top]=1;
        maxn[top]=minn[top]=a[sa[i]];
    }
    for(int i=1;i<=n;++i)sum[i]+=sum[i-1];
    for(int i=n-2;i>=0;--i)
        Max[i]=max(Max[i],Max[i+1]);
    for(int i=0;i<n;++i){
        printf("%lld ",sum[i]);
        if(Max[i]!=-INF)printf("%lld\n",Max[i]);
        else printf("0\n");
    }
    return 0;
}
```
