这题啊，直接异或第k小+二分即可。

因为能异或出的数的个数知道，找到大于K的最小的数是第几小即可知道大于K的数有多少了。

不过呢，这题数据较弱，有些细节不考虑都能过，例如可以异或出0时有部分代码会出bug。

> hack数据：  
> 2 12  
> 13 2

参考代码

```c++
#include <bits/stdc++.h>
using namespace std;
const int N=62;
long long d[N],p[N];
int cnt;
int n;
long long K;
inline void add(long long x){
    for(int i=60;i>=0;--i){
        if(x&(1ll<<i)){
            if(d[i])x^=d[i];
            else{
                d[i]=x;
                return;
            }
        }
    }
}
inline void work(){
    for(int i=60;i>=0;--i)
        for(int j=i-1;j>=0;--j)
            if(d[i]&(1ll<<j))d[i]^=d[j];
    cnt=0;
    for(int i=0;i<=60;++i)
        if(d[i])p[cnt++]=d[i];
}
long long S;
inline long long kth(long long k){
    if(k==1&&cnt<n)return 0;
    if(cnt<n)--k;
    long long ans=0;
    for(int i=cnt-1;i>=0;--i)
        if(k&(1ll<<i))ans^=p[i];
    return ans;
} 
int main(){
    scanf("%d%lld",&n,&K);
    long long x;
    for(int i=1;i<=n;++i){
        scanf("%lld",&x);
        add(x);
    }
    work();
    if(cnt<n)S=1ll<<cnt;
    else S=(1ll<<cnt)-1;
    long long L=1,R=S,ans=0;
    while(L<=R){
        long long  mid=L+R>>1;
        if(kth(mid)<=K)L=mid+1;
        else R=mid-1,ans=mid;
    }
    printf("%lld\n",S-ans+1);
    return 0;
}
```
