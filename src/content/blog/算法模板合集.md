### 1.2 FWT(快速沃尔什变换)

可用于求

$$

C[i\oplus j]=\sum_{i}\sum_{j}a[i]*b[j]

$$

逻辑运算符可以为 异或，与，或

```c++
namespace wjr {
#define ull unsigned long long
#define uint unsigned int
#define bs bitset<N>
#define lb(x) x&-x
    typedef double db;
    typedef long long LL;
    int mod = 998244353, inv2 = 499122177, inf = 1e9 + 7;
    inline void initial() {
        inv2 = (mod + 1) >> 1;
    }
    inline int mul(int a, int b) { return a * 1ll * b % mod; }
    inline int add(int a, int b) { return a += b, a >= mod ? a - mod : a; }
    inline int del(int a, int b) { return a -= b, a < 0 ? a + mod : a; }
    inline int qpow(int a, int b) { int s = 1; for (; b; b >>= 1, a = mul(a, a))if (b & 1)s = mul(s, a); return s; }
    template<typename T>
    inline T sqr(register T x) { return x * x; }
}using namespace wjr;

namespace FWT {
    const int N = 1 << 20;
    void maintain(int& n) {
        int q = 1;
        while (q < n)
            q <<= 1;
        n = q;
    }
    void FWT_and(int* a, int n, int p) {
        int step, s, i, j;
        for (step = 1, s = 2; step < n; step <<= 1, s <<= 1)
            for (i = 0; i < n; i += s)
                for (j = i; j < i + step; ++j)
                    a[j] = ~p ? add(a[j], a[j + step]) : del(a[j], a[j + step]);
    }void FWT_or(int* a, int n, int p) {
        int step, s, i, j;
        for (step = 1, s = 2; step < n; step <<= 1, s <<= 1)
            for (i = 0; i < n; i += s)
                for (j = i; j < i + step; ++j)
                    a[j + step] = ~p ? add(a[j + step], a[j]) : del(a[j + step], a[j]);
    }void FWT_xor(int* a, int n, int p) {
        int step, s, i, j, x;
        for (step = 1, s = 2; step < n; step <<= 1, s <<= 1)
            for (i = 0; i < n; i += s)
                for (j = i; j < i + step; ++j) {
                    x = a[j];
                    a[j] = add(a[j], a[j + step]);
                    a[j + step] = del(x, a[j + step]);
                    if (~p)continue;
                    a[j] = mul(a[j], inv2);
                    a[j + step] = mul(a[j + step], inv2);
                }
    }
    void FWTAND(int* a, int n, int* b, int m, int* c) {
        static int A[N], B[N];
        int len = max(n, m);
        maintain(len);
        memcpy(A, a, n << 2);
        memset(A + n, 0, (len - n) << 2);
        memcpy(B, b, m << 2);
        memset(B + m, 0, (len - m) << 2);
        FWT_and(A, len, 1);
        FWT_and(B, len, 1);
        for (int i = 0; i < len; ++i)
            A[i] = mul(A[i], B[i]);
        FWT_and(A, len, -1);
        len = max(n, m);
        memcpy(c,A,len<<2);
    }
    void FWTOR(int* a, int n, int* b, int m, int* c) {
        static int A[N], B[N];
        int len = max(n, m);
        maintain(len);
        memcpy(A, a, n << 2);
        memset(A + n, 0, (len - n) << 2);
        memcpy(B, b, m << 2);
        memset(B + m, 0, (len - m) << 2);
        FWT_or(A, len, 1);
        FWT_or(B, len, 1);
        for (int i = 0; i < len; ++i)
            A[i] = mul(A[i], B[i]);
        FWT_or(A, len, -1);
        len = max(n, m);
        memcpy(c, A, len << 2);
    }
    void FWTXOR(int* a, int n, int* b, int m, int* c) {
        static int A[N], B[N];
        int len = max(n, m);
        maintain(len);
        memcpy(A, a, n << 2);
        memset(A + n, 0, (len - n) << 2);
        memcpy(B, b, m << 2);
        memset(B + m, 0, (len - m) << 2);
        FWT_xor(A, len, 1);
        FWT_xor(B, len, 1);
        for (int i = 0; i < len; ++i)
            A[i] = mul(A[i], B[i]);
        FWT_xor(A, len, -1);
        len = max(n, m);
        memcpy(c, A, len << 2);
    }
}
```

### 1.3 FST(快速子集变换)

可用于求

$$

C[T]=\sum a[S]*b[T \oplus S]

$$

逻辑运算符为异或

也可以写成

$$

C[k]=\sum_{i\&j=0\i|j=k}a[i]*[j]

$$

```c++
namespace wjr {
#define ull unsigned long long
#define uint unsigned int
#define bs bitset<N>
#define lb(x) x&-x
    typedef double db;
    typedef long long LL;
    int mod = 998244353, inv2 = 499122177, inf = 1e9 + 7;
    inline void initial() {
        inv2 = (mod + 1) >> 1;
    }
    inline int mul(int a, int b) { return a * 1ll * b % mod; }
    inline int add(int a, int b) { return a += b, a >= mod ? a - mod : a; }
    inline int del(int a, int b) { return a -= b, a < 0 ? a + mod : a; }
    inline int qpow(int a, int b) { int s = 1; for (; b; b >>= 1, a = mul(a, a))if (b & 1)s = mul(s, a); return s; }
    template<typename T>
    inline T sqr(register T x) { return x * x; }
}using namespace wjr;

namespace FST {
    const int N = 1 << 20;
    int ctz[N];
    void initial() {
        ctz[0] = 0;
        ctz[1] = 1;
        for (int i = 2; i < N; ++i) 
            ctz[i] = ctz[i ^ (i & -i)] + 1;
    }
    void maintain(int& n) {
        int q = 1;
        while (q < n)
            q <<= 1;
        n = q;
    }
    void FST(int* a, int n, int* b, int m, int* c) {
        static int A[21][1 << 20], B[21][1 << 20], C[21][1 << 20];
        int len = max(n, m);
        maintain(len);
        int lim = 0;
        while ((1 << lim) < len)
            ++lim;
        //初始化
        for(int i=0;i<=lim;++i)
            for(int j=0;j<len;++j)
                A[i][j]=B[i][j]=0;
        for (int i = 0; i < n; ++i)
            A[ctz[i]][i] = a[i];
        for (int i = 0; i < m; ++i)
            B[ctz[i]][i] = b[i];

        for (int i = 0; i <= lim; ++i)
            FWT::FWT_or(A[i], len, 1), FWT::FWT_or(B[i], len, 1);
        for (int i = 0; i <= lim; ++i)
            for (int j = 0; j <= i; ++j)
                for (int k = 0; k < len; ++k)
                    C[i][k] = add(C[i][k], mul(A[j][k], B[i - j][k]));
        for (int i = 0; i <= lim; ++i)
            FWT::FWT_or(C[i], len, -1);
        for (int i = 0; i < (1 << lim); ++i)
            c[i] = C[ctz[i]][i];
    }
}
```

使用方法：传入a数组及其大小，b数组及其大小，以及答案数组c

### 1.4 NTT

多项式卷积（模数按需修改）

```c++
namespace wjr {
#define ull unsigned long long
#define uint unsigned int
#define bs bitset<N>
#define lb(x) x&-x
    typedef double db;
    typedef long long LL;
    int mod = 998244353, inv2 = 499122177, inf = 1e9 + 7;
    inline void initial() {
        inv2 = (mod + 1) >> 1;
    }
    inline int mul(int a, int b) { return a * 1ll * b % mod; }
    inline int add(int a, int b) { return a += b, a >= mod ? a - mod : a; }
    inline int del(int a, int b) { return a -= b, a < 0 ? a + mod : a; }
    inline int qpow(int a, int b) { int s = 1; for (; b; b >>= 1, a = mul(a, a))if (b & 1)s = mul(s, a); return s; }
    template<typename T>
    inline T sqr(register T x) { return x * x; }
}using namespace wjr;

/*---多项式 - NTT */
namespace Poly_NTT {
    const int N = 1 << 20;
    int g[25], gi[25], rev[N], INV[N | 1];
    inline void init(int n) {/*---初始化---*/
        int bit = 1;
        while ((1 << bit) < (n << 1))++bit;
        int I = qpow(3, mod - 2);
        for (int i = 1; i <= bit; ++i)g[i] = qpow(3, (mod - 1) >> i), gi[i] = qpow(I, (mod - 1) >> i);
        INV[0] = INV[1] = 1; for (int i = 2; i <= (1 << bit); ++i)INV[i] = mul(INV[mod % i], mod - mod / i);
    }inline void NTT(int* a, int n, int p) {
        int i, j, step, s, enk, wn, x, now;
        for (i = 0; i < n; ++i)if (i < rev[i])swap(a[i], a[rev[i]]);
        for (step = 1, s = 2, now = 1; step < n; step <<= 1, s <<= 1, ++now) {
            wn = ~p ? g[now] : gi[now];
            for (i = 0; i < n; i += s) {
                enk = 1;
                for (j = i; j < i + step; ++j) {
                    x = mul(enk, a[j + step]);
                    a[j + step] = del(a[j], x);
                    a[j] = add(a[j], x);
                    enk = mul(enk, wn);
                }
            }
        }if (~p)return;
        for (i = 0; i < n; ++i)a[i] = mul(a[i], INV[n]);
    }inline void get_rev(int s) { int Maxn = 1 << s; for (int i = 0; i < Maxn; ++i)rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (s - 1)); }
    inline void Mul(int* a, int n, int* b, int m) {/*---多项式乘法---*/
        static int A[N], B[N];
        int i, bit, s, len; len = n + m - 1;
        bit = 1, s = 2;
        while (s < len)++bit, s <<= 1;
        get_rev(bit);
        memcpy(A, a, n << 2), memset(A + n, 0, (s - n) << 2);
        memcpy(B, b, m << 2), memset(B + m, 0, (s - m) << 2);
        NTT(A, s, 1), NTT(B, s, 1);
        for (i = 0; i < s; ++i)A[i] = mul(A[i], B[i]);
        NTT(A, s, -1); memcpy(a, A, len << 2);
    }inline void Inv(int* a, int n) {/*---多项式逆元---*/
        static int A[N], B[N], s, i, j, k, now;
        s = 2; while (s < n)s <<= 1;
        memset(a + n, 0, (s - n) << 2);
        memset(B, 0, s << 3);
        B[0] = a[0] <= N ? INV[a[0]] : qpow(a[0], mod - 2);
        for (i = 2, j = 4, now = 2; i <= s; i <<= 1, j <<= 1, ++now) {
            memcpy(A, a, i << 2), memset(A + i, 0, i << 2);
            get_rev(now);
            NTT(A, j, 1), NTT(B, j, 1);
            for (k = 0; k < j; ++k)B[k] = mul(B[k], del(2, mul(A[k], B[k])));
            NTT(B, j, -1), memset(B + i, 0, i << 2);
        }memcpy(a, B, n << 2);
    }inline void deriv(int* a, int n) {/*---求导函数---*/
        for (int i = 0; i < n; ++i)a[i] = mul(a[i + 1], i + 1); a[n - 1] = 0;
    }inline void intag(int* a, int n) {/*---求原函数---*/
        for (int i = n - 1; i; --i)a[i] = mul(a[i - 1], INV[i]); a[0] = 0;
    }inline void Ln(int* a, int n) {/*---对数函数---*/
        static int A[N];
        memcpy(A, a, n << 2), Inv(A, n);
        deriv(a, n),
            Mul(a, n, A, n);
        intag(a, n), memset(a + n, 0, (n - 1) << 2);
    }inline void Exp(int* a, int n) {/*---指数函数---*/
        static int A[N], B[N], i, j, k, now, s;
        s = 2; while (s < n)s <<= 1;
        memset(a + n, 0, (s - n) << 2);
        memset(B, 0, s << 3);
        B[0] = 1;
        for (i = 2, j = 4, now = 2; i <= s; i <<= 1, j <<= 1, ++now) {
            memcpy(A, B, i << 2), memset(A + i, 0, i << 2);
            Ln(A, i); for (k = 0; k < i; ++k)A[k] = del(a[k], A[k]);
            A[0] = add(A[0], 1), get_rev(now);
            NTT(A, j, 1), NTT(B, j, 1);
            for (k = 0; k < j; ++k)B[k] = mul(B[k], A[k]);
            NTT(B, j, -1), memset(B + i, 0, i << 2);
        }memcpy(a, B, n << 2);
    }inline void Ksm(int* a, int n, int k) {/*---快速幂---*/
        /*---对于x^n取模，最终长度是n而不是(n-1)k---*/
        int t, p, inv;
        t = 0; while (!a[t] && t < n)++t;
        if (t >= n)return;
        p = a[t], inv = p <= N ? INV[p] : qpow(p, mod - 2);
        for (int i = 0; i < n - t; ++i)a[i] = mul(a[i + t], inv);
        memset(a + n - t, 0, t << 2);
        Ln(a, n);
        for (int i = 0; i < n; ++i)a[i] = mul(a[i], k);
        a[0] = 0, Exp(a, n);
        for (int i = n - 1; i >= t; --i)
            a[i] = mul(p, a[i - t]);
        memset(a, 0, t << 2);
    }inline void Div(int* a, int n, int* b, int m) {/*---多项式除法+取模---*/
        static int F[N], G[N];
        memcpy(F, a, (n + 1) << 2), memcpy(G, b, (m + 1) << 2);
        reverse(b, b + 1 + m), reverse(F, F + 1 + n);
        for (int i = n - m + 1; i <= m; ++i)b[i] = 0;
        Inv(b, n - m + 1);
        for (int i = n - m + 1; i <= n; ++i)F[i] = 0;
        Mul(F, n - m + 1, b, n - m + 1);
        reverse(F, F + n - m + 1);
        for (int i = n - m + 1; i <= n; ++i)F[i] = 0;
        Mul(G, m + 1, F, n - m + 1);
        for (int i = 0; i < m; ++i)b[i] = del(a[i], G[i]);
        memcpy(a, F, n << 2);
    }inline void Sqrt(int* a, int n) {//多项式开根 
        static int A[N], B[N];
        int s, i, j, k, now;
        s = 2; while (s < n)s <<= 1;
        memset(a + n, 0, (s - n) << 2), memset(B, 0, s << 3);
        B[0] = 1;
        for (i = 2, j = 4, now = 2; i <= s; i <<= 1, j <<= 1, ++now) {
            memcpy(A, B, i << 2), memset(A + i, 0, i << 2);
            for (k = 0; k < i; ++k)A[k] = mul(A[k], 2);
            Inv(A, i);
            get_rev(now - 1);
            NTT(B, i, 1);
            for (k = 0; k < i; ++k)B[k] = mul(B[k], B[k]);
            NTT(B, i, -1);
            for (k = 0; k < i; ++k)B[k] = add(B[k], a[k]);
            get_rev(now);
            NTT(B, j, 1), NTT(A, j, 1);
            for (k = 0; k < j; ++k)B[k] = mul(B[k], A[k]);
            NTT(B, j, -1), memset(B + i, 0, i << 2);
        }memcpy(a, B, n << 2);
    }
}
```

### 1.5 FFT

NTT是带取模的FFT

```c++
namespace wjr {
#define ull unsigned long long
#define uint unsigned int
#define bs bitset<N>
#define lb(x) x&-x
    typedef double db;
    typedef long long LL;
    int mod = 998244353, inv2 = 499122177, inf = 1e9 + 7;
    inline void initial() {
        inv2 = (mod + 1) >> 1;
    }
    inline int mul(int a, int b) { return a * 1ll * b % mod; }
    inline int add(int a, int b) { return a += b, a >= mod ? a - mod : a; }
    inline int del(int a, int b) { return a -= b, a < 0 ? a + mod : a; }
    inline int qpow(int a, int b) { int s = 1; for (; b; b >>= 1, a = mul(a, a))if (b & 1)s = mul(s, a); return s; }
    template<typename T>
    inline T sqr(register T x) { return x * x; }
}using namespace wjr;
namespace Poly_FFT {
    const int N = 1 << 18, zw = (1 << 15) - 1;
    const double PI = acos(-1);
    int rev[N];
    struct cd {
        double x, y;
        cd(double a = 0, double b = 0) { x = a, y = b; }
    }w[18][N];
    inline cd operator+(cd a, cd b) { return cd(a.x + b.x, a.y + b.y); }
    inline cd operator-(cd a, cd b) { return cd(a.x - b.x, a.y - b.y); }
    inline cd operator*(cd a, cd b) { return cd(a.x * b.x - a.y * b.y, a.x * b.y + a.y * b.x); }
    inline cd operator*(cd a, int b) { return cd(a.x, a.y * b); }
    inline void init(int n) {
        for (int i = 0, k = 1, j; i <= 17; ++i, k <<= 1)
            for (j = 0; j < k; ++j)
                w[i][j] = cd(cos(PI * j / k), sin(PI * j / k));
    }inline void get_rev(int s) { 
        int Maxn = 1 << s; 
        for (int i = 0; i < Maxn; ++i)rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (s - 1)); 
    }
    inline void FFT(register cd* a, int n, int p) {
        int step, s, i, j, now;
        cd x;
        for (i = 0; i < n; ++i)if (i < rev[i])swap(a[i], a[rev[i]]);
        for (step = 1, s = 2, now = 0; step < n; step <<= 1, s <<= 1, ++now)
            for (i = 0; i < n; i += s)
                for (j = i; j < i + step; ++j) {
                    x = a[j + step] * (w[now][j - i] * p);
                    a[j + step] = a[j] - x;
                    a[j] = a[j] + x;
                }
        if (~p)return;
        for (i = 0; i < n; ++i)a[i].x /= n;
    }inline LL get(cd x, int Mod) { return LL(x.x + 0.5) % Mod; }
    int siz = sizeof(cd);
    const int pw = 32767;
    inline void MTT(int* a, int n,int*b,int m,int*c, int Mod) {//对于较大模数需要用MTT
        static cd A[N], B[N], C[N], D[N], e[N], f[N], g[N], h[N];
        int i, s, len,bit; 
        len = n + m - 1;
        bit=1;
        s = 2; 
        while (s < len)s <<= 1,++bit;
        for (i = 0; i < s; ++i) {
            A[i] = cd(a[i] >> 15, 0), B[i] = cd(a[i] & pw, 0);
            C[i] = cd(b[i] >> 15, 0), D[i] = cd(b[i] & pw, 0);
        }
        get_rev(bit);
        FFT(A, s, 1), FFT(B, s, 1), FFT(C, s, 1), FFT(D, s, 1);
        for (i = 0; i < s; ++i)
            e[i] = A[i] * C[i], f[i] = A[i] * D[i], g[i] = B[i] * C[i], h[i] = B[i] * D[i];
        FFT(e, s, -1), FFT(f, s, -1), FFT(g, s, -1), FFT(h, s, -1);
        for (i = 0; i < len; ++i)
            c[i] = (((get(e[i], Mod) << 30) % Mod + (get(f[i], Mod) << 15) % Mod) % Mod + (get(g[i], Mod) << 15) % Mod + get(h[i], Mod) % Mod) % Mod;
    }inline void MTT_2(int* a, int n, int* b, int m) {//黑科技优化版本 
        static cd A[N], B[N], C[N], D[N];
        int len = n + m - 1, s = 2, i, t, bit = 1;
        register LL q, w, e, r;
        while (s < len)s <<= 1,++bit;
        for (i = 0; i < n; ++i)A[i] = cd(a[i] & zw, a[i] >> 15);
        for (i = n; i < s; ++i)A[i] = cd(0, 0);
        for (i = 0; i < m; ++i)B[i] = cd(b[i] & zw, b[i] >> 15);
        for (i = m; i < s; ++i)B[i] = cd(0, 0);
        get_rev(bit);
        FFT(A, s, 1), FFT(B, s, 1);
        for (i = 0; i < s; ++i) {
            t = (s - i) & (s - 1);
            C[i] = cd(0.5 * (A[i].x + A[t].x), 0.5 * (A[i].y - A[t].y)) * B[i];
            D[i] = cd(0.5 * (A[i].y + A[t].y), 0.5 * (A[t].x - A[i].x)) * B[i];
        }FFT(C, s, -1), FFT(D, s, -1);
        for (i = 0; i < len; ++i) {
            q = LL(C[i].x + 0.5) % mod, w = LL(C[i].y + 0.5) % mod, e = LL(D[i].x + 0.5) % mod, r = LL(D[i].y + 0.5) % mod;
            a[i] = (q + ((w + e) << 15) + (r << 30)) % mod;
```

### 1.6 BSGS

求解形如

$$

a^t \equiv b(mod \,p)

$$

给定a,b,p，其中p是质数

求解 t

复杂度为$O(\sqrt{p})$

```c++
namespace myhash {
    const unsigned int fib = 2654435769;
    inline unsigned int hash(unsigned int x) { return (x * fib) >> 14; }
}
/*---BSGS---*/
namespace BSGS {
    const int  N = 1e4 + 10;
    int d[1<<18], nxt[N], to[N], id[N], tot;
    int stk[N], tail;
    inline void Insert(int x, int i) {
        int now = myhash::hash(x);
        if (!d[now])stk[++tail] = d[now];
        to[++tot] = x, nxt[tot] = d[now], d[now] = tot, id[tot] = i;
    }inline int query(int x) {
        int now = myhash::hash(x);
        for (int i = d[now]; i; i = nxt[i])
            if (to[i] == x)return id[i];
        return -1;
    }inline void init() { while (tail)d[stk[tail--]] = 0; tot = 0; }//初始化
    inline int qpow(int a, int b, int p) { int s = 1; for (; b; b >>= 1, a = a * 1ll * a % p)if (b & 1)s = s * 1ll * a % p; return s; }
    inline int BSGS(int a, int b, int p) {//求a^t ≡b (mod p)
        if (b == 1)return 0;
        int tmp = ceil(sqrt(p));
        int t = qpow(a, tmp, p), x = 1;
        for (int i = 1; i <= tmp; ++i) {
            x = x * 1ll * t % p;
            if(query(x)==-1)Insert(x, i);
        }x = b;
        int A = qpow(a, p - 2, p);
        for (int i = 0; i < tmp; ++i) {
            t = query(x);
            if (~t)return t * tmp - i;
            x = x * 1ll * A % p;
        }return -1;
    }
}
```

### 1.8 中国剩余定理

求解形如

$$

\left \{ \begin{array}{c} x \equiv a_1\,(mod\,b_1) \\ x \equiv a_2\,(mod\,b_2)\\ \vdots \\ x \equiv a_n\,(mod\,b_n) \end{array} \right .

$$

```c++
/*---中国剩余定理---*/
namespace EXCRT {
    const int N = 1e4 + 10;
    const double eps = 1e-2;
    typedef long long LL;
    struct node {
        LL a[N], b[N];
        LL ans, M;
    };
    LL x, y;
    inline LL exgcd(LL a, LL b, LL& x, LL& y) {
        if (!b) { x = 1, y = 0; return a; }
        LL g = exgcd(b, a % b, y, x);
        y -= (a / b) * x;
        return g;
    }inline LL Mul(LL a, LL b, LL p) {
        LL d = ((long double)a / p * b + eps);
        LL r = a * b - p * d;
        return r < 0 ? r + p : r;
    }inline LL Del(LL a, LL b, LL p) { return a -= b, a < 0 ? a + p : a; }
    inline void CRT(node& a, int n) {
        a.ans = a.a[1], a.M = a.b[1];
        LL c, exg, p;
        for (int i = 2; i <= n; ++i) {//a.ans+x*a.M≡a.a[i](mod a.b[i]) 
            c = Del(a.a[i], a.ans, a.b[i]);//x*a.M≡a.a[i]-a.ans(mod a.b[i])
            exg = exgcd(a.M, a.b[i], x, y);
            p = a.b[i] / exg;
            if (c % exg != 0) { a.ans = -1; return; }
            x = Mul(x, c / exg, p);
            a.ans += x * a.M;
            a.M *= p;
            a.ans = (a.ans % a.M + a.M) % a.M;
        }
    }
}
```

### 1.11 莫比乌斯反演

1.  对于任意正整数 n ，$\sum\_{d|n}\mu(d)=[n=1]$（这个性质非常重要，后面给出证明）
2.  对于任意正整数 n , $\sum\_{d|n}{\frac{\mu(d)}{d}}=\frac{\phi(n)}{n}$（这个性质也很重要）

莫比乌斯函数求解：

```c++
int mu[maxn],pri[maxn];//prime简写
bool f[maxn];
inline void work(int n){
    mu[1]=1;f[1]=true;
    for(int i=2;i<=n;++i){
        if(!f[i]){
            pri[++pri[0]]=i;
            mu[i]=-1;
        }
        for(int j=1;j<=pri[0]&&i*pri[j]<=n;++j){
            f[i*pri[j]]=true;
            if(i%pri[j]==0)break;
            mu[i*pri[j]]=-mu[i];
        }
    }
}
```

-   定理：F(n)和f(n)是定义在非负整数集合上的两个函数，并且满足条件：

$$

F(n)=\sum_{d|n} f(d)

$$

​ 那么存在一个结论：

$$

f(n)=\sum_{d|n}\mu(d)F(\frac{n}{d})

$$

-   当然，莫比乌斯反演有另外的一种形式，当F(n)和f(n)满足：
    
$$

F(n)=\sum_{n|d}f(d)

$$
    
    可以推出
    
$$

f(n)=\sum_{n|d}\mu(\frac{d}{n})F(d)

$$
    
    通常用整除分块优化复杂度。

-   双栈模拟队列（兼容队列和双端队列）

主要用途：

栈中容易维护，但是队列/双端队列中不好维护

复杂度O(n log n \* 栈中维护复杂度)

* * *

a\[0\]是队首栈，a\[1\]是队尾栈

-   push：插入到栈 F 中。
    
-   pop：如果 S 非空，让 S 弹栈；否则将F中前一半的元素倒着插入S，F中只保留后一半，再在S中弹栈。
    

```c++
//a[0]表示队首栈，a[1]表示队尾栈
struct stk {
    int st[N],top;
}a[2];
//向队首/队尾压栈，压栈时一般需要维护栈
void push(int o,int x) {
    a[o].st[++a[o].top]=x;
}
void pop(int o) {//弹栈，兼容队列和双端队列，复杂度为O(nlogn)
    if (a[o].top) {//如果栈中有元素直接弹栈
        --a[o].top;
        return;
    }
    //否则，将另一个栈中一半的元素倒着压入该栈并弹栈
    int k=!o;
    int mid=(a[k].top+1)>>1,top=a[k].top;
    for(int i=mid;i>=1;--i)push(o,a[k].st[i]);
    a[k].top=0;
    for(int i=mid+1;i<=top;++i)push(k,a[k].st[i]);
    --a[o].top;
}
```

```c++
struct line {

        }
        return;
    }
    update(lc, L, mid, ll, rr, k);

}
```

### 2.6 吉斯林线段树

功能：

对区间取min或max，求区间最值/区间和。

* * *

线段树每个节点记录最大值，次大值，最大值个数，标记。

1.  如果 $MAX\leq t$，显然这个 是没有意义的，直接返回；
2.  如果$Se < t \leq Max$ ，那么这个 就能更新当前区间中的最大值。于是我们让区间和加上 ，然后更新 为 ，并打一个标记。
3.  如果 $t\leq Se$，那么这时你发现你不知道有多少个数涉及到更新的问题。于是我们的策略就是，暴力递归向下操作。然后上传信息。

势能分析法可得复杂度是O(mlogn)

```c++
inline void pushup(int x) {
    sum[x]=sum[lc]+sum[rc];
    if (mx[lc] == mx[rc]) {
        mx[x]=mx[lc];
        se[x]=max(se[lc],se[rc]);
        cnt[x]=cnt[lc]+cnt[rc];
    }
    else if (mx[lc] > mx[rc]) {
        mx[x]=mx[lc];
        se[x]=max(se[lc],mx[rc]);
        cnt[x]=cnt[lc];
    }
    else {
        mx[x]=mx[rc];
        se[x]=max(mx[lc],se[rc]);
        cnt[x]=cnt[rc];
    }
}
inline void pushr(int x,int v) {
    if(mx[x]<=v)return;
    sum[x]+=(v-mx[x])*1ll*cnt[x];
    tag[x]=mx[x]=v;
}inline void pushdown(int x) {
    if (tag[x] != -inf) {
        pushr(lc,tag[x]);
        pushr(rc,tag[x]);
        tag[x]=-inf;
    }
}
void build(int& x, int L, int R) {
    x=++tot;
    tag[x]=-inf;
    if (L == R) {
        mx[x]=sum[x]=b[L];
        cnt[x]=1;
        se[x]=-inf;
        return;
    }int mid=L+R>>1;
    build(lc,L,mid);
    build(rc,mid+1,R);
    pushup(x);
}
void update(int x, int L, int R, int ll, int rr, int v) {
    if(mx[x]<=v)return;
    if (ll <= L && R <= rr&&se[x]<v) return pushr(x,v);
    int mid=L+R>>1;
    pushdown(x);
    if(ll<=mid)update(lc,L,mid,ll,rr,v);
    if(rr>mid) update(rc,mid+1,R,ll,rr,v);
    pushup(x);
}
```

```c++
    inline bool cmp(register pnt a, register pnt b) { return a.x[opt] < b.x[opt]; }
    inline int f(pnt a, pnt b) { return sqr(a.x[0] - b.x[0]) + sqr(a.x[1] - b.x[1]); }
    int ch[M][2], minn[M][2], maxn[M][2], cnt;
    pnt*d[M];
    inline void pushup(const int&x) {
        for (int i = 0; i < 2; ++i)
            maxn[x][i] = minn[x][i] = d[x]->x[i];
        if (lc)
            for (int i = 0; i < 2; ++i)
                maxn[x][i] = max(maxn[x][i], maxn[lc][i]), 
                minn[x][i] = min(minn[x][i], minn[lc][i]);
        if (rc)
            for (int i = 0; i < 2; ++i)
                maxn[x][i] = max(maxn[x][i], maxn[rc][i]), 
                minn[x][i] = min(minn[x][i], minn[rc][i]);
    }inline int build(int L, int R, int k) {
        if (L > R)return 0;
        int x = ++cnt; 
        opt = k;
        int mid = (L + R) >> 1;
        nth_element(a + L, a + mid, a + R + 1, cmp);
        d[x] = &a[mid];
        lc = build(L, mid - 1, k ^ 1), rc = build(mid + 1, R, k ^ 1);
        pushup(x);
    }inline int mindis(pnt&a, int b) {
        return sqr(max(a.x[0]-maxn[b][0],0)+max(minn[b][0]-a.x[0],0))+sqr(max(a.x[1]-maxn[b][1],0)+max(minn[b][1]-a.x[1],0));
    }inline int maxdis(pnt&a, int b) {//最远距离
        return max(sqr(maxn[b][0] - a.x[0]), sqr(minn[b][0] - a.x[0])) + max(sqr(maxn[b][1] - a.x[1]), sqr(minn[b][1] - a.x[1]));
     }inline void kqueryMax(int x, pnt&O) {//k远点对查询
         if(!x)return;
        int dl = -inf, dr = -inf;
     	if (lc)dl = maxdis(O, lc);
        if (rc)dr = maxdis(O, rc);
        if (d[x] != &O) {
            int dis = f(*d[x], O);
            if (q.top().dis < dis)
                q.pop(), q.push(dis);
        }
        if (dl > dr) { 
            if (q.top().dis < dl)kqueryMax(lc, O); 
            if (q.top().dis < dr)kqueryMax(rc, O); 
        }
        else { 
            if (q.top().dis < dr)kqueryMax(rc, O); 
            if (q.top().dis < dl)kqueryMax(lc, O); 
        }
     }inline void queryMin(int x, pnt&O) {//最近距离
         if (!x)return;
         int dl = inf, dr = inf;
         if (lc)dl = mindis(O, lc);
         if (rc)dr = mindis(O, rc);
         if (d[x] != &O)//不是相同的点
             ans = min(ans, f(*d[x], O));
         if (dl < dr) {
             if (dl < ans)queryMin(lc, O);
             if (dr < ans)queryMin(rc, O);
         }
         else {
             if (dr < ans)queryMin(rc, O);
             if (dl < ans)queryMin(lc, O);
         }
     }inline void queryMax(int x, pnt& O) {//最远距离
         if (!x)return;
         int dl = -inf, dr = -inf;
         if (lc)dl = maxdis(O, lc);
         if (rc)dr = maxdis(O, rc);
         if (d[x] != &O)//不是相同的点
             ans = max(ans, f(*d[x], O));
         if (dl > dr) {
             if (ans < dl)queryMax(lc, O);
             if (ans < dr)queryMax(rc, O);
         }
         else {
             if (ans < dr)queryMax(rc, O);
             if (ans < dl)queryMax(lc, O);
         }
    }
#undef lc
#undef rc 
}
```

注意事项：

1.这是int类型的点，坐标是double的话改一下即可

2.单次查询复杂度最坏是O(n)

3.k远点对需要现在最小堆丢入2k个 0，k近点对即丢入2k个inf

### 3.2 SA

```c++
/*---SA后缀数组---*/
namespace SA {
    const int N = 1e4 + 10;
    int tax[N], sa[N], rk[N], tp[N];
    int h[N], bit[N], s[30];
    int n, m;
    int Min[N][19];
    //h为height数组,Min用于RMQ
    inline void init(int n) {//预处理 
        for (int i = 2; i <= n; ++i)bit[i] = bit[i >> 1] + 1;
        s[0] = 1;
        for (int i = 1; i <= bit[n]; ++i)s[i] = s[i - 1] << 1;
    }inline void qsort() {
        int i;
        memset(tax + 1, 0, m << 2);
        for (i = 1; i <= n; ++i)++tax[rk[i]];
        for (i = 1; i <= m; ++i)tax[i] += tax[i - 1];
        for (i = n; i; --i)sa[tax[rk[tp[i]]]--] = tp[i];
    }inline void suffix_sort(register char* s) {
        int i, j, p;
        n = strlen(s + 1), m = 26;
        for (i = 1; i <= n; ++i)rk[i] = s[i] - 'a', tp[i] = i;
        qsort();
        for (i = 1, p = 0; p < n; m = p, i <<= 1) {
            p = 0;
            for (j = 1; j <= i; ++j)tp[++p] = n - i + j;
            for (j = 1; j <= n; ++j)if (sa[j] > i)tp[++p] = sa[j] - i;
            qsort(), swap(tp, rk);
            rk[sa[1]] = p = 1;
            for (j = 2; j <= n; ++j)
                rk[sa[j]] = (tp[sa[j]] == tp[sa[j - 1]] && tp[sa[j] + i] == tp[sa[j - 1] + i]) ? p : ++p;
        }
    }inline void work(register char* s) {
        suffix_sort(s);
        int i, j, k = 0;
        for (i = 1; i <= n; ++i) {
            if (k)--k;
            j = sa[rk[i] - 1];
            while (s[i + k] == s[j + k])++k;
            h[rk[i]] = k;
        }for (i = 1; i <= n; ++i)Min[i][0] = h[i];
        for (j = 1; j <= bit[n]; ++j) {
            int p = s[j - 1], q = s[j];
            for (i = 1; i <= n - q + 1; ++i)
                Min[i][j] = min(Min[i][j - 1], Min[i + p][j - 1]);
        }
    }
}
```

### 3.3 Manacher

求出一个字符串每个位置的最长回文子串

$p[i]$表示从$i$ 向两侧延伸（当然要保证两侧对应位置相等）的最大长度。

为了处理方便，我们把原串每两个字符之间加一个（不包含在原串中的）`#`，开头加一个 `$`。

这样得到的回文串长度就保证是奇数了

```c++
void Manacher() {
    int mx=0,id;
    for (int i = 0; i<n ; ++i) {
        if(mx>i)p[i]=min(p[id*2-i],p[id]+id-i);
        else p[i]=1;
        while(i>=p[i]&&s[i+p[i]]==s[i-p[i]])++p[i];
        if (p[i] + i > mx) {
            mx=p[i]+i;
            id=i;
        }
    }
}
```

### 4.2 模拟退火

对于答案更优的就接受，对于答案更劣的概率接受

假设当前温度为Temp，则接受的概率为$e^{\frac{\Delta f}{Temp}}(\Delta f<0)$

其中有几个需要注意的点：

-   初始点的选取对算法结果有一定的影响，最好是多次运行对结果进行综合判断。
-   在算法运行初期，温度下降快，避免接受过多的差结果。当运行时间增加，温度下降减缓，以便于更快稳定结果。
-   当迭代次数增加到一定次数时，结果可能已经达到稳定，但是距离算法结束还有一段时间。在设计程序时应该加入适当的输出条件，满足输出条件即可结束程序。

降温可以每次乘以一个$tmp,tmp\in [0.95,0.99]$

或者自定义温度函数以使得算法更优。

### 5.1 多重背包

用二进制分组优化

例如：

-   6=1+2+3
-   8=1+2+4+1
-   18=1+2+4+8+3
-   31=1+2+4+8+16

对于$k\_i+1$不是2的整次幂，最后需要添加一个$k\_i-2^{\lfloor log\_2(k\_i+1)\rfloor}+1$的物品

### 5.4 分组背包

所谓分组背包，就是将物品分组，每组的物品相互冲突，最多只能选一个物品放进去。

这种题怎么想呢？其实是从「在所有物品中选择一件」变成了「从当前组中选择一件」，于是就对每一组进行一次 0-1 背包就可以了。

```c++
for (int k = 1; k <= ts; k++)          // 循环每一组
  for (int i = m; i >= 0; i--)         // 循环背包容量
    for (int j = 1; j <= cnt[k]; j++)  // 循环该组的每一个物品
      if (i >= w[t[k][j]])
        dp[i] = max(dp[i],dp[i - w[t[k][j]]] + c[t[k][j]]);  // 像0-1背包一样状态转移
```

即对于每一组只能取0/1件，因此从大到小枚举背包容量
